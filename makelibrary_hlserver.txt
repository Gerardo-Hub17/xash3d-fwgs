target_compile_options(hl PRIVATE -fpermissive -fno-exceptions -MMD -MP -fno-threadsafe-statics -Wno-invalid-offsetof -fvisibility=hidden
-fno-rtti
#-fno-lto
-fno-unwind-tables -fno-asynchronous-unwind-tables
# -G0
)

	CXX_VISIBILITY_PRESET hidden
    VISIBILITY_INLINES_HIDDEN YES
)
