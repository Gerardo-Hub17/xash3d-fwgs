# El cliente compila sus propias CBasePlayerWeapon, CBaseEntity, CAK47, pm_shared...
# con el mismo nombre que las del servidor (hl = ReGameDLL) pero con otro contenido
# (#ifdef CLIENT_DLL) y otro layout. Con --allow-multiple-definition el enlazador las
# mezclaria y el servidor acabaria llamando a las versiones del cliente. Se renombra todo
# simbolo global de la libreria del cliente salvo lib_cl_dll_exports (la tabla que lee el motor).
if(NOT CMAKE_NM OR NOT CMAKE_OBJCOPY)
	message(FATAL_ERROR "cl_isolate: hacen falta nm y objcopy del toolchain (CMAKE_NM=${CMAKE_NM} CMAKE_OBJCOPY=${CMAKE_OBJCOPY})")
endif()
add_custom_command(TARGET cl_dll POST_BUILD
	COMMAND sh ${CMAKE_CURRENT_SOURCE_DIR}/scripts/cl_isolate.sh ${CMAKE_NM} ${CMAKE_OBJCOPY} $<TARGET_FILE:cl_dll>
	VERBATIM
)

target_compile_definitions(hl PRIVATE
	REGAMEDLL_FIXES REGAMEDLL_API REGAMEDLL_ADD UNICODE_FIXES BUILD_LATEST
	CLIENT_WEAPONS USE_QSTRING
	_LINUX LINUX NDEBUG
	BUILD_STATIC_LIBSTDC     # no incluir stdc++compat.cpp (trucos de glibc)
	GEKKO HW_RVL             # ramas de Wii en osconfig.h / interface.* / FileSystem.cpp
	_stricmp=strcasecmp _strnicmp=strncasecmp _strdup=strdup _unlink=unlink
	_snprintf=snprintf _vsnprintf=vsnprintf _write=write _close=close
	_access=access _vsnwprintf=vswprintf
)

# -fsigned-char: en PowerPC "char" es sin signo y ReGameDLL (x86) da por hecho que lleva signo.
# No se usa -fno-rtti: cbase.cpp tiene dynamic_cast.
target_compile_options(hl PRIVATE
	$<$<COMPILE_LANGUAGE:CXX>:-std=gnu++14>
	-fpermissive -fno-exceptions -fno-threadsafe-statics -fsigned-char
	-ffunction-sections -fdata-sections -fvisibility=hidden
	-fno-unwind-tables -fno-asynchronous-unwind-tables
	-Wno-delete-non-virtual-dtor -Wno-invalid-offsetof -Wno-write-strings
	-Wno-unused-variable -Wno-unused-value -Wno-unused-function -Wno-switch
	-Wno-sign-compare -Wno-format -Wno-strict-aliasing
)

set_target_properties(hl PROPERTIES
	CXX_VISIBILITY_PRESET hidden
	VISIBILITY_INLINES_HIDDEN YES
