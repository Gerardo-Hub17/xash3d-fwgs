)
target_compile_definitions(hl PRIVATE -DXASH_OGC -Dstricmp=strcasecmp -Dstrnicmp=strncasecmp -D_snprintf=snprintf  -DNO_VOICEGAMEMGR)
target_compile_options(hl PRIVATE -fpermissive -fno-exceptions -MMD -MP -fno-threadsafe-statics -Wno-invalid-offsetof -fvisibility=hidden
-fno-rtti
#-fno-lto
-fno-unwind-tables -fno-asynchronous-unwind-tables
# -G0
)
set_target_properties(hl PROPERTIES
	CXX_VISIBILITY_PRESET hidden
    VISIBILITY_INLINES_HIDDEN YES
)

target_include_directories(hl PRIVATE
	${CMAKE_CURRENT_SOURCE_DIR}/../hlsdk-portable/dlls
